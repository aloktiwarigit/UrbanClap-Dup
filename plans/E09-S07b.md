```typescript
import { BROWSER_API_BASE_URL } from './base';
import type { AdminTechnician, AdminTechnicianListResponse, PatchTechnicianBody } from '@/types/technician-admin';
import type { ApiClient } from './client';

const BASE = BROWSER_API_BASE_URL;

export async function listTechnicians(client: ApiClient): Promise<AdminTechnicianListResponse> {
  const { data, error } = await client.GET('/v1/admin/technicians' as never, {});
  if (error !== undefined || data === undefined) throw new Error('listTechnicians: request failed');
  return data as AdminTechnicianListResponse;
}

export async function patchTechnician(id: string, body: PatchTechnicianBody): Promise<void> {
  const res = await fetch(`${BASE}/v1/admin/technicians/${encodeURIComponent(id)}`, {
    method: 'PATCH',
    headers: { 'content-type': 'application/json' },
    credentials: 'include',
    body: JSON.stringify(body),
  });
  if (!res.ok) throw new Error(`patchTechnician: ${res.status}`);
}
```

```typescript
'use client';

import { useState, useMemo } from 'react';
import { useTranslations } from 'next-intl';
import { useAdminAuth } from '@/lib/auth/context';
import { patchTechnician } from '@/api/technicians';
import type { AdminTechnician, TechnicianStatus, PatchTechnicianBody } from '@/types/technician-admin';

interface Props {
  initialTechnicians: AdminTechnician[];
}

type StatusFilter = 'ALL' | TechnicianStatus;

const STATUS_COLORS: Record<TechnicianStatus, string> = {
  ON_DUTY:   'var(--teal)',
  OFF_DUTY:  'var(--fog-0)',
  SUSPENDED: 'var(--rose)',
};

const KYC_COLORS = {
  VERIFIED: 'var(--teal)',
  PENDING:  'var(--ember)',
  REJECTED: 'var(--rose)',
};

export function TechnicianRosterClient({ initialTechnicians }: Props) {
  const t = useTranslations('technicians');
  const { auth } = useAdminAuth();
  const isSuperAdmin = auth?.role === 'super-admin';

  const [technicians, setTechnicians] = useState(initialTechnicians);
  const [search, setSearch] = useState('');
  const [statusFilter, setStatusFilter] = useState<StatusFilter>('ALL');
  const [editingCommission, setEditingCommission] = useState<string | null>(null);
  const [commissionDraft, setCommissionDraft] = useState('');
  const [loading, setLoading] = useState<string | null>(null);

  const filtered = useMemo(() => {
    return technicians.filter((tech) => {
      const matchesSearch =
        !search ||
        tech.name.toLowerCase().includes(search.toLowerCase()) ||
        tech.phone.includes(search);
      const matchesStatus = statusFilter === 'ALL' || tech.status === statusFilter;
      return matchesSearch && matchesStatus;
    });

  async function handleStatusChange(id: string, newStatus: TechnicianStatus) {
    setLoading(id);
    try {
      await patchTechnician(id, { status: newStatus });
      setTechnicians((prev) =>
        prev.map((t) => (t.id === id ? { ...t, status: newStatus } : t)),
      );
    } finally {
      setLoading(null);
    }
  }

    const pct = parseInt(commissionDraft, 10);
    if (isNaN(pct) || pct < 0 || pct > 100) return;

    try {
      await patchTechnician(id, { commissionPct: pct });
      setTechnicians((prev) =>
        prev.map((t) => (t.id === id ? { ...t, commissionPct: pct } : t)),

      setLoading(null);

            value={search}

              color: 'var(--fog-2)', fontSize: '0.8125rem', minWidth: '220px',

          />

            value={statusFilter}
            onChange={(e) => setStatusFilter(e.target.value as StatusFilter)}
            aria-label="Filter by status"

              padding: '6px 10px', borderRadius: 'var(--radius-sm)',
```

```typescript
import { BROWSER_API_BASE_URL } from './base';
import type { AdminCustomer, AdminCustomerListResponse, CustomerStatus } from '@/types/customer-admin';
import type { ApiClient } from './client';

const BASE = BROWSER_API_BASE_URL;

export async function listCustomers(client: ApiClient): Promise<AdminCustomerListResponse> {
  const { data, error } = await client.GET('/v1/admin/customers' as never, {});
  if (error !== undefined || data === undefined) throw new Error('listCustomers: request failed');
  return data as AdminCustomerListResponse;
}

export async function patchCustomer(id: string, accountStatus: CustomerStatus): Promise<void> {
  const res = await fetch(`${BASE}/v1/admin/customers/${encodeURIComponent(id)}`, {
    method: 'PATCH',
    headers: { 'content-type': 'application/json' },
    credentials: 'include',
    body: JSON.stringify({ accountStatus }),
  });
  if (!res.ok) throw new Error(`patchCustomer: ${res.status}`);
}

export async function addCustomerNote(id: string, text: string): Promise<void> {
  const res = await fetch(`${BASE}/v1/admin/customers/${encodeURIComponent(id)}/notes`, {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    credentials: 'include',
    body: JSON.stringify({ text }),
  });
  if (!res.ok) throw new Error(`addCustomerNote: ${res.status}`);
}

export async function refundCredit(id: string, amountRupees: number, reason: string): Promise<void> {
  const res = await fetch(`${BASE}/v1/admin/customers/${encodeURIComponent(id)}/refund-credit`, {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    credentials: 'include',
    body: JSON.stringify({ amountRupees, reason }),
  });
  if (!res.ok) throw new Error(`refundCredit: ${res.status}`);
}
```

```typescript
'use client';

import { useState, useMemo } from 'react';
import { useTranslations } from 'next-intl';
import { patchCustomer, addCustomerNote, refundCredit } from '@/api/customers';
import type { AdminCustomer, CustomerStatus } from '@/types/customer-admin';

interface Props {
  initialCustomers: AdminCustomer[];
}

type StatusFilter = 'ALL' | CustomerStatus;

function RelativeDate({ iso }: { iso?: string }) {
  if (!iso) return <span style={{ color: 'var(--fog-0)' }}>—</span>;
  const days = Math.floor((Date.now() - new Date(iso).getTime()) / 86400000);
  if (days === 0) return <span>Today</span>;
  if (days === 1) return <span>Yesterday</span>;
  return <span>{days} days ago</span>;
}

export function CustomerListClient({ initialCustomers }: Props) {
  const t = useTranslations('customers');

  const [customers, setCustomers] = useState(initialCustomers);
  const [search, setSearch] = useState('');
  const [statusFilter, setStatusFilter] = useState<StatusFilter>('ALL');
  const [expanded, setExpanded] = useState<string | null>(null);
  const [loading, setLoading] = useState<string | null>(null);
  const [noteText, setNoteText] = useState<Record<string, string>>({});
  const [showNoteInput, setShowNoteInput] = useState<string | null>(null);
  const [showRefundForm, setShowRefundForm] = useState<string | null>(null);
  const [refundAmount, setRefundAmount] = useState('');
  const [refundReason, setRefundReason] = useState('');

  const filtered = useMemo(() => {
    return customers.filter((c) => {
      const matchesSearch =
        !search ||
        c.name.toLowerCase().includes(search.toLowerCase()) ||
        c.phone.includes(search);
      const matchesStatus = statusFilter === 'ALL' || c.accountStatus === statusFilter;
      return matchesSearch && matchesStatus;
    });
  }, [customers, search, statusFilter]);

  async function handleFlag(id: string, current: CustomerStatus) {
    const newStatus: CustomerStatus = current === 'FLAGGED' ? 'ACTIVE' : 'FLAGGED';
    setLoading(id);
    try {
      await patchCustomer(id, newStatus);
      setCustomers((prev) =>
        prev.map((c) => (c.id === id ? { ...c, accountStatus: newStatus } : c)),
      );
    } finally {
      setLoading(null);
    }
  }

  async function handleAddNote(id: string) {
    const text = noteText[id]?.trim();
    if (!text) return;
    setLoading(id);
    try {
      await addCustomerNote(id, text);
      const now = new Date().toISOString();
      setCustomers((prev) =>
        prev.map((c) =>
          c.id === id
            ? { ...c, notes: [...c.notes, { text, createdAt: now, authorName: 'admin' }] }
            : c,
        ),
      );
      setNoteText((prev) => ({ ...prev, [id]: '' }));
      setShowNoteInput(null);
    } finally {
      setLoading(null);
    }
  }

  async function handleRefundCredit(id: string) {
    const amount = parseInt(refundAmount, 10);
    if (isNaN(amount) || amount <= 0 || !refundReason.trim()) return;
    setLoading(id);
    try {
      await refundCredit(id, amount, refundReason.trim());
      setShowRefundForm(null);
      setRefundAmount('');
      setRefundReason('');
    } finally {
      setLoading(null);
    }
  }

  return (
    <div style={{ padding: '1.5rem', display: 'flex', flexDirection: 'column', gap: '1rem' }}>
      {/* Header */}
      <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', flexWrap: 'wrap', gap: '0.75rem' }}>
        <h1 style={{ margin: 0, fontFamily: 'var(--font-display)', fontSize: '1.5rem', color: 'var(--fog-3)' }}>
          {t('title')}
        </h1>
        <div style={{ display: 'flex', gap: '0.5rem', flexWrap: 'wrap' }}>
          <input
            type="search"
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            placeholder={t('searchPlaceholder')}
            aria-label={t('searchPlaceholder')}
            style={{
              padding: '6px 10px', borderRadius: 'var(--radius-sm)',
              border: '1px solid var(--ink-4)', background: 'var(--ink-2)',
              color: 'var(--fog-2)', fontSize: '0.8125rem', minWidth: '220px',
            }}
          />
          <select
            value={statusFilter}
            onChange={(e) => setStatusFilter(e.target.value as StatusFilter)}
            aria-label="Filter by status"
            style={{
              padding: '6px 10px', borderRadius: 'var(--radius-sm)',
              border: '1px solid var(--ink-4)', background: 'var(--ink-2)',
              color: 'var(--fog-2)', fontSize: '0.8125rem',
            }}
          >
            <option value="ALL">{t('filterAll')}</option>
            <option value="ACTIVE">{t('status.ACTIVE')}</option>
            <option value="FLAGGED">{t('status.FLAGGED')}</option>
          </select>
        </div>
      </div>

      {/* Table */}
      {filtered.length === 0 ? (
        <p style={{ color: 'var(--fog-0)', textAlign: 'center', padding: '3rem 0' }}>
          No customers found.
        </p>
      ) : (
        <div style={{ overflowX: 'auto' }}>
          <table style={{ width: '100%', borderCollapse: 'collapse', fontSize: '0.8125rem' }}>
            <thead>
              <tr style={{ borderBottom: '1px solid var(--ink-4)', color: 'var(--fog-0)', textAlign: 'left' }}>
                <th style={{ padding: '8px 12px' }}>{t('columns.name')}</th>
                <th style={{ padding: '8px 12px' }}>{t('columns.phone')}</th>
                <th style={{ padding: '8px 12px' }}>{t('columns.city')}</th>
                <th style={{ padding: '8px 12px' }}>{t('columns.bookings')}</th>
                <th style={{ padding: '8px 12px' }}>{t('columns.lastBooking')}</th>
                <th style={{ padding: '8px 12px' }}>{t('columns.openComplaints')}</th>
                <th style={{ padding: '8px 12px' }}>{t('columns.status')}</th>
                <th style={{ padding: '8px 12px' }}>{t('columns.actions')}</th>
              </tr>
            </thead>
            <tbody>
              {filtered.map((customer) => (
                <>
                  <tr
                    key={customer.id}
                    style={{ borderBottom: expanded === customer.id ? 'none' : '1px solid var(--ink-3)', cursor: 'pointer' }}
                    onClick={() => setExpanded(expanded === customer.id ? null : customer.id)}
                  >
                    {/* Name */}
                    <td style={{ padding: '10px 12px', color: 'var(--fog-3)', fontWeight: 500 }}>
                      {customer.name}
                      <span style={{ marginLeft: 6, color: 'var(--fog-0)', fontSize: '0.7rem' }}>
                        {expanded === customer.id ? '▲' : '▼'}
                      </span>
                    </td>

                    {/* Phone */}
                    <td style={{ padding: '10px 12px', color: 'var(--fog-1)', fontFamily: 'var(--font-mono)', fontSize: '0.75rem' }}>
                      {customer.phone}
                    </td>

                    {/* City */}
                    <td style={{ padding: '10px 12px', color: 'var(--fog-1)' }}>
                      {customer.city}
                    </td>

                    {/* Bookings */}
                    <td style={{ padding: '10px 12px', textAlign: 'center', color: 'var(--fog-2)' }}>
                      {customer.bookingCount}
                    </td>

                    {/* Last booking */}
                    <td style={{ padding: '10px 12px', color: 'var(--fog-1)', fontSize: '0.75rem' }}>
                      <RelativeDate iso={customer.lastBookingDate} />
                    </td>

                    {/* Open complaints */}
                    <td style={{ padding: '10px 12px', textAlign: 'center' }}>
                      <span style={{
                        color: customer.openComplaintCount > 0 ? 'var(--rose)' : 'var(--fog-0)',
                        fontWeight: customer.openComplaintCount > 0 ? 600 : 400,
                      }}>
                        {customer.openComplaintCount}
                      </span>
                    </td>

                    {/* Status */}
                    <td style={{ padding: '10px 12px' }}>
                      <span style={{
                        padding: '3px 8px', borderRadius: '4px', fontSize: '0.75rem', fontWeight: 600,
                        background: customer.accountStatus === 'FLAGGED'
                          ? 'color-mix(in srgb, var(--rose) 15%, transparent)'
                          : 'color-mix(in srgb, var(--teal) 15%, transparent)',
                        color: customer.accountStatus === 'FLAGGED' ? 'var(--rose)' : 'var(--teal)',
                      }}>
                        {t(`status.${customer.accountStatus}` as 'status.ACTIVE')}
                      </span>
                    </td>

                    {/* Actions */}
                    <td style={{ padding: '10px 12px' }} onClick={(e) => e.stopPropagation()}>
                      <div style={{ display: 'flex', gap: '6px', flexWrap: 'wrap' }}>
                        <button
                          onClick={() => void handleFlag(customer.id, customer.accountStatus)}
                          disabled={loading === customer.id}
                          style={{
                            padding: '4px 8px', borderRadius: 3, border: 'none', cursor: 'pointer', fontSize: '0.75rem',
                            background: customer.accountStatus === 'FLAGGED'
                              ? 'var(--ink-3)' : 'color-mix(in srgb, var(--rose) 15%, transparent)',
                            color: customer.accountStatus === 'FLAGGED' ? 'var(--fog-1)' : 'var(--rose)',
                          }}
                        >
                          {customer.accountStatus === 'FLAGGED' ? t('actions.unflag') : t('actions.flag')}
                        </button>
                        <button
                          onClick={() => setShowNoteInput(showNoteInput === customer.id ? null : customer.id)}
                          style={{ padding: '4px 8px', borderRadius: 3, border: '1px solid var(--ink-4)', background: 'transparent', color: 'var(--fog-1)', cursor: 'pointer', fontSize: '0.75rem' }}
                        >
                          {t('actions.addNote')}
                        </button>
                        <button
                          onClick={() => setShowRefundForm(showRefundForm === customer.id ? null : customer.id)}
                          style={{ padding: '4px 8px', borderRadius: 3, border: '1px solid var(--marigold-dim)', background: 'transparent', color: 'var(--marigold)', cursor: 'pointer', fontSize: '0.75rem' }}
                        >
                          {t('actions.refundCredit')}
                        </button>
                      </div>

                      {/* Inline note input */}
                      {showNoteInput === customer.id && (
                        <div style={{ marginTop: 8, display: 'flex', gap: 4 }}>
                          <input
                            type="text"
                            value={noteText[customer.id] ?? ''}
                            onChange={(e) => setNoteText((prev) => ({ ...prev, [customer.id]: e.target.value }))}
                            placeholder="Add note..."
                            style={{ flex: 1, padding: '4px 8px', background: 'var(--ink-2)', border: '1px solid var(--ink-4)', color: 'var(--fog-2)', borderRadius: 3, fontSize: '0.8rem' }}
                          />
                          <button
                            onClick={() => void handleAddNote(customer.id)}
                            disabled={loading === customer.id}
                            style={{ padding: '4px 10px', background: 'var(--teal)', color: 'var(--ink-0)', border: 'none', borderRadius: 3, cursor: 'pointer', fontSize: '0.75rem' }}
                          >
                            Save
                          </button>
                        </div>
                      )}

                      {/* Inline refund form */}
                      {showRefundForm === customer.id && (
                        <div style={{ marginTop: 8, display: 'flex', flexDirection: 'column', gap: 4, background: 'var(--ink-2)', padding: 10, borderRadius: 4 }}>
                          <input
                            type="number"
                            value={refundAmount}
                            onChange={(e) => setRefundAmount(e.target.value)}
                            placeholder="Amount (₹)"
                            min={1} max={10000}
                            style={{ padding: '4px 8px', background: 'var(--ink-3)', border: '1px solid var(--ink-4)', color: 'var(--fog-2)', borderRadius: 3, fontSize: '0.8rem' }}

                            style={{ padding: '4px 8px', background: 'var(--ink-3)', border: '1px solid var(--ink-4)', color: 'var(--fog-2)', borderRadius: 3, fontSize: '0.8rem', resize: 'vertical' }}
                          />
                          <button
                            onClick={() => void handleRefundCredit(customer.id)}
                            disabled={loading === customer.id}
                            style={{ padding: '4px 10px', background: 'var(--marigold)', color: 'var(--ink-0)', border: 'none', borderRadius: 3, cursor: 'pointer', fontSize: '0.75rem', fontWeight: 600 }}
                          >
                            Issue Refund Credit
                          </button>

                    </td>
                  </tr>

                  {/* Expanded row */}
                  {expanded === customer.id && (
                    <tr key={`${customer.id}-expanded`}>
                      <td colSpan={8} style={{ padding: '0 12px 16px 24px', borderBottom: '1px solid var(--ink-4)' }}>
                        <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr 1fr', gap: '1.5rem', paddingTop: 12 }}>
                          {/* Recent bookings */}

                            <p style={{ margin: '0 0 8px', color: 'var(--fog-0)', fontSize: '0.7rem', textTransform: 'uppercase', letterSpacing: '0.08em' }}>
                              {t('expandedSection.recentBookings')}
                            </p>
                            {customer.recentBookings.length === 0 ? (
                              <p style={{ color: 'var(--fog-0)', fontSize: '0.75rem' }}>—</p>
                            ) : customer.recentBookings.map((b, i) => (
                              <div key={i} style={{ fontSize: '0.75rem', color: 'var(--fog-1)', marginBottom: 4 }}>
                                <span style={{ color: 'var(--fog-0)' }}>{b.date}</span> · {b.service} · {b.status}

                            <p style={{ margin: '0 0 8px', color: 'var(--fog-0)', fontSize: '0.7rem', textTransform: 'uppercase', letterSpacing: '0.08em' }}>
```
